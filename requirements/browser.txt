# What runs in the browser. Mirrors what ovos-installer puts on a device
# (ansible/roles/ovos_virtualenv/templates/virtualenv/*-requirements.txt.j2),
# minus the parts that are hardware: listener, audio service, PHAL.
# Built with `pip wheel --pre`, i.e. the alpha channel.

# core-requirements.txt.j2: ovos-core[plugins], public speech (the installer's default)
ovos-core[plugins]
ovos-stt-plugin-server
ovos-tts-plugin-server

# skills-requirements.txt.j2 (profile "ovos"): essential, internet and audio skills
ovos-core[skills-essential,skills-internet,skills-audio]

# audio-requirements.txt.j2: ovos-audio, for its dialog and TTS transformer
# services only (speech.py); playback is the page's
ovos-audio

# GUI (installer feature "gui"): ovos-gui + homescreen (see browser-nodeps.txt).
# ovos-gui is held at the last 1.5 alpha: 1.6.0a1 moved clients to adapter
# plugins and the legacy websocket adapter ovos-visage speaks is not on PyPI yet.
ovos-gui==1.5.4a1

# Emkay's own additions to the "ovos" profile. A phone's browser cannot rely
# on servers as a device can (public STT/TTS/translate servers fail, and many
# sites do not allow a browser to read them, CORS), so skills that answer
# offline come as standard:
ovos-skill-calculator
ovos-skill-geography
ovos-skill-geometry
ovos-skill-nato-alphabet
ovos-skill-morse-code
# "tell me a joke" is among the first things people try (the default skills
# have no jokes); works in a browser (the store's test)
ovos-skill-icanhazdadjokes
# not ovos-skill-convert (yet): in Danish its "{x} i {y}" intent takes
# "hvilken dag er det i dag", and its 210 unit names make padaos' warm-up
# the longest of all; installable from the control panel
# reading aloud (ovos-common-reading-pipeline-plugin) and the providers whose
# sources a browser may read (they send CORS headers). The fairy tale
# providers (andersenstories.com, grimmstories.com) do not, so they wait.
ovos-common-reading-pipeline-plugin
ovos-skill-ovosblog
ovos-skill-horoscope-readings
# something to play: OCP's media providers whose sources a browser may read
# and play (tried in Emkay on 2026-11-17: "play jazz radio", "play the
# news"); radio-browser stations through pyradios, and the news bulletins
ovos-media-provider-pyradios
ovos-media-provider-news

# short, current facts ("who is the prime minister of denmark", "where is the
# eiffel tower") from Wikidata and OpenStreetMap, which a browser may read -
# in place of DuckDuckGo and Wolfram Alpha, which it may not (boot.py)
ovos-skill-wikidata

# who the assistant is, as Emkay: takes the place of ovos-skill-personal,
# which boot.py blacklists (REPLACED_SKILLS)
ovos-skill-about-emkay
# what went out and who hears you, from Emkay's network log (docs/network-log.md),
# and "go offline" (docs/local-only.md)
ovos-skill-guardian
# find and install skills by voice, from the skill store(s) in its settings
ovos-skill-store
# "full screen", "keep the screen on", "install as an app" (ovos-phal-plugin-browser does it)
ovos-skill-kiosk
# Home Assistant's Assist in the intent pipeline ("turn off the light on the
# stairs"), placed after adapt high by pipeline_place.py. It stays silent
# until the Home Assistant link (Settings > brain, "Connect to my Home
# Assistant") says HA is there, so without HA it costs nothing
ovos-assist-pipeline-plugin

# the setup page: the stock OpenVoiceOS control panel
ovos-control-panel

# Emkay's own browser plugins, each its own package (emkay-voice/<name>):
# the browser's speech recognition, Piper voices in WebAssembly, the
# browser tab as the device (PHAL), the page's <audio> element as
# ovos-audio's player for music, radio and podcasts, and Whisper speech
# recognition on the phone (offline)
ovos-stt-plugin-browser
ovos-tts-plugin-piper-browser>=0.1.0a3
ovos-phal-plugin-browser
ovos-audio-plugin-browser
ovos-stt-plugin-whisper-browser

# browser glue
pyodide-http

# Emkay Dev (?dev, py/emkay_mobile/dev.py): a skill's golden utterances are
# judged by the same judge as ovos-tui-client and the store's CI
ovos-routing-judge
